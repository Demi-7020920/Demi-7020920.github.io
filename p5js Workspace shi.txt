function setup() {
  createCanvas(400, 400);
  rectMode(CENTER);
  angleMode(DEGREES);
}

function draw() {
  background(220);
  translate(200,200);
  for(let i = 0; i <= 360; i+= 10){// 45,90,135
    push();
    rotate(i);
    rect(0, 0, 50, 30);
    console.log(i);
    pop();
  }
  //rotate(45);

  circle(0,0,10); //origin
}
